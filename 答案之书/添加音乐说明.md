## 📁 添加音乐文件步骤

### 步骤 1：创建音乐文件夹
在答案之书目录下创建 `music` 文件夹：
```
答案之书/
├── index.html
├── style.css
├── script.js
└── music/           ← 新建文件夹
    └── background.mp3  ← 放置音乐文件
```

### 步骤 2：准备音乐文件
支持的音乐格式：
- ✅ **MP3**（推荐，兼容性最好）
- ✅ **OGG**（开源格式）
- ✅ **WAV**（无损，文件较大）

### 步骤 3：修改 index.html

找到以下代码：
```html
<audio id="bgMusic" loop preload="auto">
    <!-- 在这里添加你的音乐文件 -->
</audio>
```

修改为：
```html
<audio id="bgMusic" loop preload="auto">
    <source src="music/background.mp3" type="audio/mpeg">
    <source src="music/background.ogg" type="audio/ogg">
    <!-- 可以添加多个格式作为备选 -->
</audio>
```

### 步骤 4：修改 script.js

找到 `createBackgroundMusic()` 函数中的这部分：
```javascript
// 初始化音乐控制
function initMusicControl() {
    musicPlayer = createBackgroundMusic();
    
    if (isMusicEnabled && musicPlayer) {
        musicPlayer.start();
        musicControl.classList.add('playing');
        musicIcon.textContent = '🎵';
    } else {
        musicControl.classList.add('muted');
        musicIcon.textContent = '🔇';
    }
}
```

替换为：
```javascript
// 初始化音乐控制（使用外部音乐文件）
function initMusicControl() {
    // 如果有音乐文件，使用外部音乐
    if (bgMusic.querySelector('source')) {
        bgMusic.volume = 0.3; // 设置音量 30%
        
        if (isMusicEnabled) {
            bgMusic.play().then(() => {
                musicControl.classList.add('playing');
                musicIcon.textContent = '🎵';
            }).catch(err => {
                console.log('音乐播放失败:', err);
                // 如果外部音乐失败，使用合成音乐
                useSynthMusic();
            });
        } else {
            musicControl.classList.add('muted');
            musicIcon.textContent = '🔇';
        }
    } else {
        // 没有外部音乐，使用合成音乐
        useSynthMusic();
    }
}

// 使用合成音乐
function useSynthMusic() {
    musicPlayer = createBackgroundMusic();
    if (isMusicEnabled && musicPlayer) {
        musicPlayer.start();
        musicControl.classList.add('playing');
        musicIcon.textContent = '🎵';
    } else {
        musicControl.classList.add('muted');
        musicIcon.textContent = '🔇';
    }
}

// 切换音乐播放
function toggleMusic() {
    // 如果使用外部音乐
    if (bgMusic.querySelector('source')) {
        if (bgMusic.paused) {
            bgMusic.play();
            musicControl.classList.add('playing');
            musicControl.classList.remove('muted');
            musicIcon.textContent = '🎵';
            isMusicEnabled = true;
            localStorage.setItem('musicEnabled', 'true');
            showToast('🎵 背景音乐已开启');
        } else {
            bgMusic.pause();
            musicControl.classList.remove('playing');
            musicControl.classList.add('muted');
            musicIcon.textContent = '🔇';
            isMusicEnabled = false;
            localStorage.setItem('musicEnabled', 'false');
            showToast('🔇 背景音乐已关闭');
        }
    } else {
        // 使用合成音乐
        if (!musicPlayer) {
            musicPlayer = createBackgroundMusic();
        }
        
        if (isMusicPlaying) {
            musicPlayer.stop();
            musicControl.classList.remove('playing');
            musicControl.classList.add('muted');
            musicIcon.textContent = '🔇';
            isMusicEnabled = false;
            localStorage.setItem('musicEnabled', 'false');
            showToast('🔇 背景音乐已关闭');
        } else {
            musicPlayer.start();
            musicControl.classList.add('playing');
            musicControl.classList.remove('muted');
            musicIcon.textContent = '🎵';
            isMusicEnabled = true;
            localStorage.setItem('musicEnabled', 'true');
            showToast('🎵 背景音乐已开启');
        }
    }
}
```

## 🎵 音乐文件优化建议

### 文件大小
- 建议：2-5MB
- 时长：2-3分钟循环播放最佳
- 比特率：128-192kbps 即可

### 音量处理
建议在代码中设置音量：
```javascript
bgMusic.volume = 0.3; // 30% 音量，不会太吵
```

### 淡入淡出
可以添加淡入淡出效果：
```javascript
// 淡入
let volume = 0;
const fadeIn = setInterval(() => {
    if (volume < 0.3) {
        volume += 0.01;
        bgMusic.volume = volume;
    } else {
        clearInterval(fadeIn);
    }
}, 50);
```

## 🔧 完整代码示例

如果你已经有合法的音乐文件，这里是完整的实现方案。

### HTML 部分
```html
<audio id="bgMusic" loop preload="auto">
    <source src="music/peaceful-piano.mp3" type="audio/mpeg">
    <source src="music/peaceful-piano.ogg" type="audio/ogg">
</audio>
```

### JavaScript 部分
```javascript
// 初始化时尝试播放外部音乐
window.addEventListener('load', () => {
    if (bgMusic.querySelector('source')) {
        // 有外部音乐文件
        bgMusic.volume = 0.3;
        
        // 用户交互后才能自动播放
        document.body.addEventListener('click', function initPlay() {
            if (isMusicEnabled) {
                bgMusic.play();
                musicControl.classList.add('playing');
            }
            document.body.removeEventListener('click', initPlay);
        }, { once: true });
    }
});
```
